The root cause of Groq 429 errors on Vercel is that
serverless functions are stateless — cooldown state
resets on every cold start so round-robin doesn't work.

The fix is: try ALL keys sequentially within a single
request until one succeeds. No cooldown state needed.

STEP 1 — Rewrite groq-keys.ts completely:

Replace the entire file with this:

const keys = (process.env.GROQ_API_KEY || '')
  .split(',')
  .map(k => k.trim())
  .filter(Boolean);

export async function makeGroqRequest(body: object): Promise<Response> {
  if (keys.length === 0) {
    throw new Error('No GROQ API keys configured');
  }

  // Shuffle keys randomly so different invocations
  // don't all start with the same key
  const shuffled = [...keys].sort(() => Math.random() - 0.5);

  let lastError: Error = new Error('All keys failed');

  for (const key of shuffled) {
    try {
      const res = await fetch(
        'https://api.groq.com/openai/v1/chat/completions',
        {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${key}`,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify(body),
          signal: AbortSignal.timeout(30000)
        }
      );

      // If rate limited, try next key immediately
      if (res.status === 429) {
        console.log(`Key ${key.substring(0,8)}... rate limited, trying next`);
        continue;
      }

      // Any other error status, try next key
      if (!res.ok && res.status >= 500) {
        continue;
      }

      // Success or client error (400 etc) — return as-is
      return res;

    } catch (err) {
      lastError = err as Error;
      console.log(`Key ${key.substring(0,8)}... failed: ${lastError.message}`);
      continue;
    }
  }

  // All keys exhausted
  throw lastError;
}

// Keep this for backwards compatibility if anything imports it
export function getNextGroqKey(): string {
  return keys[Math.floor(Math.random() * keys.length)] || '';
}

import { makeGroqRequest } from '../utils/groqClient.js';
// or wherever groq-keys.ts is located

const response = await makeGroqRequest({
  model: 'llama-3.3-70b-versatile',
  messages: [...],  // keep existing messages
  max_tokens: 1000,
  temperature: 0.8
});

if (!response.ok) {
  const errData = await response.json().catch(() => ({}));
  return res.status(response.status).json({
    error: response.status === 429
      ? 'Rate limit reached. Please wait and try again.'
      : 'AI service temporarily unavailable.'
  });
}
