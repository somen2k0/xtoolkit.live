import { Router } from 'express';
const router = Router();

// Use whichever fetch method worked in Step 1
// Direct: fetch('https://api.guerrillamail.com/ajax.php?...')
// Proxy:  fetch('https://api.allorigins.win/raw?url=' + encodeURIComponent('https://api.guerrillamail.com/ajax.php?...'))

const GUERRILLA_BASE = 'https://api.guerrillamail.com/ajax.php';
const HEADERS = {
  'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
  'Referer': 'https://www.guerrillamail.com/',
  'Accept': 'application/json'
};

function buildUrl(params: string): string {
  // Replace this with proxy wrapper if direct failed in Step 1
  return `${GUERRILLA_BASE}?${params}`;
}

// GET /api/guerrilla/new
router.get('/new', async (req, res) => {
  try {
    const r = await fetch(buildUrl('f=get_email_address&lang=en'), {
      headers: HEADERS,
      signal: AbortSignal.timeout(10000)
    });
    const data = await r.json();
    res.json(data);
  } catch (e: any) {
    res.status(500).json({ error: e.message });
  }
});

// POST /api/guerrilla/set-user
router.post('/set-user', async (req, res) => {
  try {
    const { user, sid_token } = req.body;
    const r = await fetch(
      buildUrl(`f=set_email_user&email_user=${encodeURIComponent(user)}&sid_token=${sid_token}&lang=en`),
      { headers: HEADERS, signal: AbortSignal.timeout(10000) }
    );
    const data = await r.json();
    res.json(data);
  } catch (e: any) {
    res.status(500).json({ error: e.message });
  }
});

// GET /api/guerrilla/inbox?sid_token=X&seq=0
router.get('/inbox', async (req, res) => {
  try {
    const { sid_token, seq } = req.query;
    const r = await fetch(
      buildUrl(`f=get_email_list&offset=0&sid_token=${sid_token}&seq=${seq || 0}`),
      { headers: HEADERS, signal: AbortSignal.timeout(10000) }
    );
    const data = await r.json();
    res.json(data);
  } catch (e: any) {
    res.status(500).json({ error: e.message });
  }
});

// GET /api/guerrilla/message/:id?sid_token=X
router.get('/message/:id', async (req, res) => {
  try {
    const { sid_token } = req.query;
    const r = await fetch(
      buildUrl(`f=fetch_email&email_id=${req.params.id}&sid_token=${sid_token}`),
      { headers: HEADERS, signal: AbortSignal.timeout(10000) }
    );
    const data = await r.json();
    res.json(data);
  } catch (e: any) {
    res.status(500).json({ error: e.message });
  }
});

export default router;

function randomName(): string {
  const f = FIRST_NAMES[Math.floor(Math.random()*FIRST_NAMES.length)];
  const l = LAST_NAMES[Math.floor(Math.random()*LAST_NAMES.length)];
  const n = Math.floor(Math.random()*99);
  return `${f}.${l}${n > 0 ? n : ''}`;
}
