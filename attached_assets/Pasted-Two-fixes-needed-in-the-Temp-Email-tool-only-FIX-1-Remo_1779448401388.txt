FIX 2 — Fix "Could not create inbox" error

The error happens because the frontend is calling
createInbox() but something fails silently.

The issue is the order of operations. Fix createInbox()
to follow this exact sequence:

async function createInbox() {
  setIsLoading(true);
  setError(null);
  try {
    // Step 1: Get new session from GuerrillaMail
    const newRes = await fetch('/api/guerrilla/new');
    if (!newRes.ok) throw new Error('Failed to get session');
    const newData = await newRes.json();
    
    const token = newData.sid_token;
    if (!token) throw new Error('No session token received');
    
    // Step 2: Set realistic username
    const name = randomName();
    const setRes = await fetch('/api/guerrilla/set-user', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ 
        user: name, 
        sid_token: token,
        domain: selectedDomain
      })
    });
    if (!setRes.ok) throw new Error('Failed to set username');
    const setData = await setRes.json();
    
    // Step 3: Update state with real email
    const emailAddr = setData.email_addr || 
                      `${name}@${selectedDomain}`;
    setSidToken(token);
    setEmail(emailAddr);
    setMessages([]);
    setSelectedMessage(null);
    
  } catch (e: any) {
    setError('Could not create inbox. Please try again.');
    console.error('createInbox error:', e.message);
  } finally {
    setIsLoading(false);
  }
}
