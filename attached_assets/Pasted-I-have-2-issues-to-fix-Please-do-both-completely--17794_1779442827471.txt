import { Router } from 'express';
const router = Router();

const PROXIES = [
  (url: string) => `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`,
  (url: string) => `https://corsproxy.io/?${encodeURIComponent(url)}`,
];

async function fetchWithFallback(url: string, options?: any) {
  for (const proxy of PROXIES) {
    try {
      const res = await fetch(proxy(url), {
        ...options,
        signal: AbortSignal.timeout(8000)
      });
      if (res.ok) return await res.json();
    } catch (e) { continue; }
  }
  throw new Error('All proxies failed');
}

// Get new inbox
router.get('/new', async (req, res) => {
  try {
    const data = await fetchWithFallback(
      'https://www.harakirimail.com/api/v1/inbox/new'
    );
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'Harakirimail unavailable' });
  }
});

// Get inbox messages
router.get('/inbox', async (req, res) => {
  try {
    const { token } = req.query;
    const data = await fetchWithFallback(
      `https://www.harakirimail.com/api/v1/inbox?token=${token}`
    );
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'Harakirimail unavailable' });
  }
});

// Get single message
router.get('/message/:id', async (req, res) => {
  try {
    const { token } = req.query;
    const data = await fetchWithFallback(
      `https://www.harakirimail.com/api/v1/inbox/${req.params.id}?token=${token}`
    );
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'Harakirimail unavailable' });
  }
});

export default router;

const keys = (process.env.GROQ_API_KEY || '')
  .split(',')
  .map(k => k.trim())
  .filter(Boolean);

let currentIndex = 0;
const cooldowns = new Set<number>();

export function getNextGroqKey(): string {
  if (keys.length === 0) throw new Error('No GROQ API keys configured');
  
  // Find next available key not in cooldown
  for (let i = 0; i < keys.length; i++) {
    const idx = (currentIndex + i) % keys.length;
    if (!cooldowns.has(idx)) {
      currentIndex = (idx + 1) % keys.length;
      return keys[idx];
    }
  }
  
  // All keys in cooldown, use first anyway
  currentIndex = (currentIndex + 1) % keys.length;
  return keys[currentIndex];
}

export function markKeyCooldown(key: string, ms = 60000) {
  const idx = keys.indexOf(key);
  if (idx === -1) return;
  cooldowns.add(idx);
  setTimeout(() => cooldowns.delete(idx), ms);
}

export function makeGroqRequest(body: object) {
  const key = getNextGroqKey();
  return fetch('https://api.groq.com/openai/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${key}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(body)
  }).then(async res => {
    if (res.status === 429) {
      markKeyCooldown(key);
      // Try next key immediately
      const nextKey = getNextGroqKey();
      return fetch('https://api.groq.com/openai/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${nextKey}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(body)
      });
    }
    return res;
  });
}
